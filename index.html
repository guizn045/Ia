async function enviarMensagem(textoUsuario) { // Adiciona a mensagem do utilizador ao array de histórico (messages) mensagensHistorico.push({ role: "user", content: textoUsuario }); try { const resposta = await fetch('/api/chat', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ messages: mensagensHistorico }) }); const dados = await resposta.json(); // Exibe a resposta da IA no teu chat adicionarMensagemNaTela("assistant", dados.reply); // Guarda também no histórico mensagensHistorico.push({ role: "assistant", content: dados.reply }); } catch (erro) { console.error("Erro:", erro); adicionarMensagemNaTela("assistant", "Erro de ligação com o servidor."); } }
